    let currentPart = null; // 'part1' or 'part2'

    let selectedBottle = null; // 1, 2, or 3

   

    // Track chemicals added to each bottle (in mL)

    let bottles = {

      1: { epsom: 0, water: 0, ammonia: 0 },

      2: { epsom: 0, water: 0, ammonia: 0 },

      3: { epsom: 0, water: 0, ammonia: 0 }

    };

   

    const TABLESPOON_ML = 14.787; // 1 tablespoon = 14.787 mL

    const THIRD_CUP_ML = 78.863;  // 1/3 cup = 78.863 mL

    // Calculate flask appearance based on chemicals

    function calculateFlaskColor(bottleNum) {

      const bottle = bottles[bottleNum];

      const epsomVolume = bottle.epsom;

      const ammoniaVolume = bottle.ammonia;

      const waterVolume = bottle.water;

     

      const totalVolume = epsomVolume + ammoniaVolume + waterVolume;

     

      if (totalVolume === 0) {

        return { hex: '#e0f2fe', name: 'Empty' };

      }

     

      // If there's Epsom salt and ammonia, show precipitate forming (cloudy white)

      if (epsomVolume > 0 && ammoniaVolume > 0) {

        // More ammonia = more precipitate = cloudier/whiter

        const reactionRatio = Math.min(ammoniaVolume / epsomVolume, 2);

        const cloudiness = Math.min(reactionRatio * 0.5, 0.9);

        const whiteValue = Math.floor(200 + (255 - 200) * cloudiness);

        const hex = `#${whiteValue.toString(16).padStart(2, '0')}${whiteValue.toString(16).padStart(2, '0')}${whiteValue.toString(16).padStart(2, '0')}`;

        return { hex, name: 'Cloudy White (Mg(OH)₂)' };

      }

     

      // Just Epsom salt (slightly cloudy)

      if (epsomVolume > 0) {

        return { hex: '#f0f9ff', name: 'Epsom Salt solution' };

      }

     

      // Just ammonia (clear with slight yellow tint)

      if (ammoniaVolume > 0) {

        return { hex: '#fffbeb', name: 'Ammonia solution' };

      }

     

      // Just water

      return { hex: '#e0f2fe', name: 'Water' };

    }

   

    function selectBottle(bottleNum) {

      selectedBottle = bottleNum;

     

      // Update UI

      document.querySelectorAll('.bottle-selector-btn').forEach(btn => {

        if (btn.dataset.bottle === String(bottleNum)) {

          btn.style.fontWeight = 'bold';

          btn.style.borderWidth = '3px';

        } else {

          btn.style.fontWeight = 'normal';

          btn.style.borderWidth = '2px';

        }

      });

     

      document.getElementById('selected-bottle-text').textContent = `Selected: Bottle ${bottleNum}`;

     

      // Enable chemical buttons

      document.querySelectorAll('.add-chemical-btn').forEach(btn => {

        btn.disabled = false;

      });

     

      updateStatus(`Bottle ${bottleNum} selected. Add chemicals using the buttons above.`);

    }

   

    function addChemicalToBottle(chemical, amount) {

      if (!selectedBottle) {

        updateStatus('Please select a bottle first!');

        return;

      }

     

      const volumeML = amount === 'tablespoon' ? TABLESPOON_ML : THIRD_CUP_ML;

      bottles[selectedBottle][chemical] += volumeML;

     

      updateBottleDisplay(selectedBottle);

     

      const amountText = amount === 'tablespoon' ? '1 Tbsp' : '1/3 Cup';

      const chemicalNames = {

        epsom: 'Epsom Salt',

        water: 'Water',

        ammonia: 'Ammonia'

      };

     

      const flaskColor = calculateFlaskColor(selectedBottle);

      updateStatus(`Added ${amountText} of ${chemicalNames[chemical]} to Bottle ${selectedBottle}. ${flaskColor.name}`);

    }

   

    function updateBottleDisplay(bottleNum) {

      const bottle = bottles[bottleNum];

     

      // Update text amounts

      document.getElementById(`bottle${bottleNum}-epsom`).textContent = bottle.epsom.toFixed(1) + ' mL';

      document.getElementById(`bottle${bottleNum}-water`).textContent = bottle.water.toFixed(1) + ' mL';

      document.getElementById(`bottle${bottleNum}-ammonia`).textContent = bottle.ammonia.toFixed(1) + ' mL';

     

      const total = bottle.epsom + bottle.water + bottle.ammonia;

      document.getElementById(`bottle${bottleNum}-total`).textContent = total.toFixed(1) + ' mL';

     

      // Update bottle amounts text below each bottle

      if (total === 0) {

        document.getElementById(`bottle${bottleNum}-amounts`).textContent = 'Empty';

      } else {

        document.getElementById(`bottle${bottleNum}-amounts`).textContent =

          `E:${bottle.epsom.toFixed(0)} W:${bottle.water.toFixed(0)} A:${bottle.ammonia.toFixed(0)}`;

      }

     

      // Update bottle liquid level and color

      const maxHeight = 140;

      const maxVolume = 500;

      const liquidHeight = Math.min((total / maxVolume) * maxHeight, maxHeight);

      const liquidY = 170 - liquidHeight;

     

      const liquidElement = document.getElementById(`bottle${bottleNum}-liquid`);

      liquidElement.setAttribute('y', liquidY);

      liquidElement.setAttribute('height', liquidHeight);

     

      // Update color

      const flaskColor = calculateFlaskColor(bottleNum);

      liquidElement.setAttribute('fill', flaskColor.hex);

    }

    function updateStatus(message) {

      document.getElementById('status-text').textContent = message;

    }

   

    function selectPart(part) {

      currentPart = part;

     

      // Update button styles

      document.querySelectorAll('.experiment-part-btn').forEach(btn => {

        btn.style.fontWeight = 'normal';

        btn.style.borderWidth = '2px';

      selectedBtn.style.borderWidth = '3px';

     

      // Update description

      const descEl = document.getElementById('experiment-description');

        descEl.textContent = 'Part 1: Vary the amount of ammonia with constant Epsom salt (1 Tbsp per bottle). Observe how ammonia amount affects precipitate formation.';

      } else {

        descEl.textContent = 'Part 2: Vary the amount of Epsom salt with constant ammonia (1/3 cup per bottle). Observe how Epsom salt amount affects precipitate formation.';

      }

     

      updateStatus(`${part === 'part1' ? 'Part 1' : 'Part 2'} selected. Select a bottle to begin adding chemicals.`);

    }

    function resetExperiment() {

      currentPart = null;

      selectedBottle = null;

     

      // Reset all bottles

      bottles = {

      };

     

      // Update displays

      for (let i = 1; i <= 3; i++) {

        updateBottleDisplay(i);

      }

     

      // Reset button styles

      document.querySelectorAll('.experiment-part-btn').forEach(btn => {

        btn.style.fontWeight = 'normal';

      });

     

      document.querySelectorAll('.bottle-selector-btn').forEach(btn => {

        btn.style.fontWeight = 'normal';

        btn.style.borderWidth = '2px';

      });

     

      document.getElementById('selected-bottle-text').textContent = 'No bottle selected';

      document.getElementById('experiment-description').textContent = '';

     

      // Disable chemical buttons

      document.querySelectorAll('.add-chemical-btn').forEach(btn => {

        btn.disabled = true;

      });

     

      updateStatus('Choose Part 1 or Part 2 to begin the experiment.');

    }

        }

      });
